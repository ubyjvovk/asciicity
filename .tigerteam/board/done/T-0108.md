### Regen script (throwaway, not committed) — `/tmp/regen-tokyo-terrain.mjs`

```js
// T-0108 — throwaway terrain regeneration for public/data/tokyo/index.json.
// Uses the cached SRTM tile only (offline). Guards against any drift beyond
// {terrain, waterLevels}; on any mismatch, aborts without touching disk.
import { existsSync, readFileSync, renameSync, writeFileSync } from 'node:fs';
import { gunzipSync } from 'node:zlib';

import { Dem, buildTerrain, decodeHgt, hgtTileName } from 'file:///home/d/asciicity/.tigerteam/worktrees/T-0108/scripts/dem.mjs';

const INDEX_PATH = '/home/d/asciicity/.tigerteam/worktrees/T-0108/public/data/tokyo/index.json';
const CACHE_DIR = '/home/d/asciicity/.tigerteam/worktrees/T-0108/.cache/dem';

const BBOX = [139.692, 35.645, 139.820, 35.715];
const ORIGIN = { lat: 35.6812, lon: 139.7671 };
const STEP = 20;

const before = JSON.parse(readFileSync(INDEX_PATH, 'utf8'));
if (before.terrain.cols !== 583 || before.terrain.rows !== 391) {
  throw new Error(`unexpected committed dims: ${before.terrain.cols}x${before.terrain.rows}`);
}

const [minLon, minLat, maxLon, maxLat] = BBOX;
const tiles = new Map();
for (let lat = Math.floor(minLat); lat <= Math.floor(maxLat); lat++) {
  for (let lon = Math.floor(minLon); lon <= Math.floor(maxLon); lon++) {
    const name = hgtTileName(lat + 0.5, lon + 0.5);
    const file = `${CACHE_DIR}/${name}.hgt.gz`;
    if (!existsSync(file)) throw new Error(`OFFLINE guard tripped: ${file}`);
    const { side, samples } = decodeHgt(gunzipSync(readFileSync(file)));
    tiles.set(name, { side, samples });
  }
}
const dem = new Dem(tiles, { bareEarth: true });

const waterRings = before.water;
if (!Array.isArray(waterRings)) throw new Error('index.water missing');

const built = buildTerrain({ bbox: BBOX, origin: ORIGIN, dem, step: STEP, waterRings });

const t = built.terrain;
if (t.cols !== before.terrain.cols || t.rows !== before.terrain.rows ||
    t.x0 !== before.terrain.x0 || t.z0 !== before.terrain.z0 || t.step !== before.terrain.step) {
  throw new Error(`terrain grid geometry drifted`);
}

const after = { ...before, terrain: built.terrain, waterLevels: built.waterLevels };
const eq = (a, b) => JSON.stringify(a) === JSON.stringify(b);
const beforeKeys = Object.keys(before).sort();
const afterKeys = Object.keys(after).sort();
if (JSON.stringify(beforeKeys) !== JSON.stringify(afterKeys)) throw new Error('key set drifted');
const changed = beforeKeys.filter((k) => !eq(before[k], after[k]));
if (JSON.stringify(changed.sort()) !== JSON.stringify(['terrain', 'waterLevels'])) {
  throw new Error(`unexpected changed key set: ${changed.join(',')}`);
}

const tmp = `${INDEX_PATH}.tmp`;
writeFileSync(tmp, JSON.stringify(after));
renameSync(tmp, INDEX_PATH);
```

const idx = JSON.parse(readFileSync('/home/d/asciicity/.tigerteam/worktrees/T-0108/public/data/tokyo/index.json','utf8'));
const { x0, z0, step, cols, rows, datum, heights } = idx.terrain;
function nearestAsl(x, z) {
  const c = Math.round((x - x0) / step);
  const r = Math.round((z - z0) / step);
  if (c < 0 || c >= cols || r < 0 || r >= rows) throw new Error(`OOB: ${x},${z}`);
  return heights[r * cols + c] + datum;
}
function transect(name, cx, cz, axis) {
  const vals = [];
  for (let i = -5; i <= 5; i++) {
    const dx = axis === 'x' ? i * 40 : 0;
    const dz = axis === 'z' ? i * 40 : 0;
    vals.push(nearestAsl(cx + dx, cz + dz));
  }
  console.log(`${name}: ${vals.map(v => v.toFixed(1)).join(' ')}`);
  return vals;
}
